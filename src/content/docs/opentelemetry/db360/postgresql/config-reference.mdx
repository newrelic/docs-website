This reference provides comprehensive YAML configurations for PostgreSQL monitoring with the New Relic Distribution of OpenTelemetry (NRDOT) Collector. These full-feature configurations include the complete PostgreSQL metric catalog for production environments across self-hosted and Amazon RDS deployments.

<Callout variant="important">
  **Looking to get started quickly?** These are advanced configurations for production use. If you're setting up PostgreSQL monitoring for the first time, we recommend starting with the standard configuration in our setup guides and returning to these comprehensive configurations once you've verified basic functionality.
</Callout>

## Self-hosted [#hosted]

<CollapserGroup>

<Collapser
  id="hosted-full"
  title="For self-hosted PostgreSQL">

This configuration enables the complete PostgreSQL metric catalog the `nrpostgresql` receiver can produce, without adding host-level metrics.

  <OtelConfig
    inputOptions={[
      {
        name: 'host',
        label: 'PostgreSQL Hostname/IP',
        codeLine: 3,
        defaultValue: 'postgresql.example.com',
        toolTip: 'Enter your PostgreSQL hostname or IP address',
      },
      {
        name: 'port',
        label: 'PostgreSQL Port',
        codeLine: 3,
        defaultValue: '5432',
        toolTip: 'Enter your PostgreSQL port number. The default value is 5432.',
      },
      {
        name: 'username',
        label: 'Database Username',
        codeLine: 4,
        defaultValue: 'newrelic',
        toolTip: 'Enter your database username for authentication',
      },
      {
        name: 'password',
        label: 'Database Password',
        codeLine: 5,
        defaultValue: 'YOUR_PASSWORD',
        toolTip: 'Enter the password you set for the monitoring user',
      },
      {
        name: 'database',
        label: 'Database Name',
        codeLine: 7,
        defaultValue: 'YOUR_DATABASE_NAME',
        toolTip: 'Enter the name of the database you want to monitor',
      },
      {
        name: 'endpoint',
        label: 'New Relic OTLP Endpoint',
        codeLine: 82,
        defaultValue: 'https://otlp.nr-data.net:4318',
        toolTip: 'Enter your New Relic OTLP endpoint. For more information, refer to https://docs.newrelic.com/docs/opentelemetry/best-practices/opentelemetry-otlp/',
      },
      {
        name: 'licenseKey',
        label: 'New Relic License Key',
        codeLine: 84,
        defaultValue: 'YOUR_LICENSE_KEY',
        toolTip: 'Enter your New Relic license key',
      },
    ]}
    config={`receivers:
  nrpostgresql:
    endpoint: "<input1>:<input2>"
    username: "<input3>"
    password: "<input4>"
    databases:
      - <input5>
    collection_interval: 15s
    events:
      db.server.top_query:
        enabled: true
      db.server.query_sample:
        enabled: true
      db.server.query_plan:
        enabled: true
    top_query_collection:
      max_rows_per_query: 1000
      top_n_query: 200
      collection_interval: 60s
      allowed_comment_keys: [nr_service_guid]
    query_sample_collection:
      max_rows_per_query: 1000
      allowed_comment_keys: [nr_service_guid]

    resource_attributes:
      db.system.version:
        enabled: true
    # All metrics disabled by default in the receiver, enabled here for full dashboard coverage
    metrics:
      postgresql.blks_hit:
        enabled: true
      postgresql.blks_read:
        enabled: true
      postgresql.database.locks:
        enabled: true
      postgresql.deadlocks:
        enabled: true
      postgresql.function.calls:
        enabled: true
      postgresql.query.conflicts:
        enabled: true
      postgresql.query.execution.time:
        enabled: true
      postgresql.sequential_scans:
        enabled: true
      postgresql.temp.io:
        enabled: true
      postgresql.temp_files:
        enabled: true
      postgresql.tup_deleted:
        enabled: true
      postgresql.tup_fetched:
        enabled: true
      postgresql.tup_inserted:
        enabled: true
      postgresql.tup_returned:
        enabled: true
      postgresql.tup_updated:
        enabled: true
      postgresql.vector.insert.duration:
        enabled: true
      postgresql.vector.insert.rows:
        enabled: true
      postgresql.vector.search.calls:
        enabled: true
      postgresql.vector.search.duration:
        enabled: true
      postgresql.vector.search.rows_returned:
        enabled: true
      postgresql.wal.delay:
        enabled: true

processors:
  batch:
  batch/metrics:
    send_batch_size: 8192
    send_batch_max_size: 8192
    timeout: 10s

exporters:
  otlp/newrelic:
    endpoint: "<input6>"
    headers:
      api-key: "<input7>"
    compression: gzip
    sending_queue:
      enabled: true
      sizer: bytes
      queue_size: 100_000_000
      num_consumers: 10
      batch:
        sizer: bytes
        max_size: 1_000_000
        min_size: 0
        flush_timeout: 5s
    retry_on_failure:
      enabled: true
      initial_interval: 5s
      max_interval: 30s
      max_elapsed_time: 300s

service:
  telemetry:
    metrics:
      level: none
  pipelines:
    metrics:
      receivers: [nrpostgresql]
      processors: [batch/metrics]
      exporters: [otlp/newrelic]
    logs:
      receivers: [nrpostgresql]
      processors: [batch]
      exporters: [otlp/newrelic]`}
    fileName="postgresql-config.yaml"
    tipMdx={{ body: null }}
    onChange={() => {}}
  />

<Callout variant="important">
PostgreSQL integration via NRDOT Collector follows standard [data ingest pricing](/docs/accounts/accounts-billing/new-relic-one-pricing-billing/data-ingest-billing/#data). Note that full feature configuration ingests more data than a standard setup, which may increase your overall bill.
</Callout>
