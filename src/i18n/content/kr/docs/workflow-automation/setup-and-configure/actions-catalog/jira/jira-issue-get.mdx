              ```yaml
              name: get-jira-issue-details
              description: 'Retrieves the summary and status of a specific Jira issue.'
              steps:
                - name: get_issue
                  type: action
                  action: jira.issue.get
                  version: 1
                  inputs:
                    authToken: ${{ :secrets:jiraAuthToken }}
                    domain: mycompany
                    issueKey: PROJ-567
                    parameters:
                      fields: "summary,status,issuetype"
                  next: end
              ```
